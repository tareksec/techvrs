export function FlixPromptWebApp() {
  const [activeCategory, setActiveCategory] = useState("animation");
  const [searchQuery, setSearchQuery] = useState("");
  const [activeTab, setActiveTab] = useState<"movie" | "series" | "originals">("movie");
  const [copiedId, setCopiedId] = useState<string | null>(null);
  const [selectedPrompt, setSelectedPrompt] = useState<PromptItem | null>(null);
  const [bookmarkedIds, setBookmarkedIds] = useState<Set<string>>(new Set());

  const copyPromptText = (id: string, text: string, e?: React.MouseEvent) => {
    if (e) e.stopPropagation();
    navigator.clipboard.writeText(text);
    setCopiedId(id);
    setTimeout(() => setCopiedId(null), 2400);
  };

  const toggleBookmark = (id: string, e?: React.MouseEvent) => {
    if (e) e.stopPropagation();
    setBookmarkedIds((prev) => {
      const next = new Set(prev);
      if (next.has(id)) next.delete(id);
      else next.add(id);
      return next;
    });
  };

  const filteredItems = useMemo(() => {
    return PROMPT_COLLECTION.filter((item) => {
      if (activeTab === "series" && !["animation", "drakor", "romance"].includes(item.category)) {
        return false;
      }
      if (activeTab === "originals" && item.category !== "special") {
        return false;
      }

      const matchesCategory =
        activeCategory === "trending" ||
        item.category.toLowerCase() === activeCategory.toLowerCase();

      const query = searchQuery.trim().toLowerCase();
      const matchesQuery =
        !query ||
        item.title.toLowerCase().includes(query) ||
        item.description.toLowerCase().includes(query) ||
        item.model.toLowerCase().includes(query) ||
        item.prompt.toLowerCase().includes(query) ||
        item.tags.some((t) => t.toLowerCase().includes(query));

      return matchesCategory && matchesQuery;
    });
  }, [activeCategory, searchQuery, activeTab]);

  return (
    <div className="min-h-screen w-full bg-[#52606e] text-slate-100 flex flex-col justify-between select-none">
      
      {/* ── Outer Matte Glass Background ── */}
      <div className="w-full flex-1 flex flex-col items-center p-3 sm:p-6 lg:p-10 bg-[radial-gradient(ellipse_at_top,_var(--tw-gradient-stops))] from-[#5e6f80] via-[#4e5c6a] to-[#414d59]">
        
        {/* Main Frosted Glass Card Container (Matching screenshot) */}
        <div className="w-full max-w-[1320px] bg-[#647485]/80 backdrop-blur-3xl rounded-[34px] sm:rounded-[42px] border border-white/20 shadow-[0_30px_90px_-15px_rgba(0,0,0,0.35)] p-6 sm:p-8 lg:p-10 relative overflow-hidden flex-1 flex flex-col">
          
          {/* Subtle Ambient Radial Lighting */}
          <div className="pointer-events-none absolute -top-40 -left-40 w-[450px] h-[450px] bg-white/10 rounded-full blur-[100px]" />
          <div className="pointer-events-none absolute -bottom-40 -right-40 w-[450px] h-[450px] bg-cyan-400/10 rounded-full blur-[100px]" />

          {/* ── 1. Top Bar Header (User Profile Removed) ── */}
          <header className="relative z-10 flex flex-col md:flex-row items-center justify-between gap-5 pb-6 border-b border-white/15">
            
            {/* Left: Logo */}
            <div className="flex items-center gap-2">
              <span className="text-2xl sm:text-3xl font-bold tracking-tight text-white drop-shadow-sm">
                Flix.id
              </span>
            </div>

            {/* Center: Skiper96 Expandable Tabs Navigation [Movie | Series | Originals | 🔍] */}
            <Skiper96
              tabs={[
                { id: "movie", title: "Movie", icon: Film },
                { id: "series", title: "Series", icon: Tv },
                { id: "originals", title: "Originals", icon: Sparkles },
              ]}
              activeTab={activeTab}
              onTabChange={(id) => setActiveTab(id as "movie" | "series" | "originals")}
            >
              {/* Inline Search Capsule */}
              <div className="relative flex items-center pl-2 pr-2">
                <input
                  type="text"
                  value={searchQuery}
                  onChange={(e) => setSearchQuery(e.target.value)}
                  placeholder="Search prompts..."
                  className="bg-transparent text-xs sm:text-sm text-white placeholder:text-white/40 focus:outline-none w-24 sm:w-36 lg:w-44 px-2"
                />
                <div className="w-6 h-6 rounded-full bg-white/5 hover:bg-white/15 flex items-center justify-center text-white/70 transition shrink-0 cursor-pointer">
                  <Search className="w-3.5 h-3.5" />
                </div>
              </div>
            </Skiper96>

            {/* Right: Clean Exit / Return to Main Website Link (User Profile Removed) */}
            <div className="flex items-center gap-3">
              <a
                href="/"
                title="Return to Main Website"
                className="flex items-center gap-2 px-4 py-2 rounded-full bg-white/15 hover:bg-white/25 border border-white/25 text-xs font-semibold text-white transition shadow-sm backdrop-blur-md"
              >
                <ArrowLeft className="w-3.5 h-3.5" />
                <span>Main Website</span>
              </a>
            </div>
          </header>

          {/* ── 2. Hero Featured Banners (Two Large Side-by-Side Cards) ── */}
          <section className="relative z-10 grid grid-cols-1 lg:grid-cols-2 gap-5 my-6 sm:my-7">
            {FEATURED_HEROES.map((hero) => (
              <div
                key={hero.id}
                className="relative h-[270px] sm:h-[300px] rounded-[26px] overflow-hidden border border-white/15 shadow-xl group cursor-pointer"
                onClick={() => {
                  setSelectedPrompt({
                    id: hero.id,
                    title: hero.title.replace(/\n/g, " "),
                    category: "Featured",
                    rating: "9.5",
                    year: "2024",
                    model: hero.model,
                    image: hero.image,
                    description: hero.subtitle,
                    prompt: hero.prompt,
                    tags: ["Featured", "Cinematic", "Masterpiece"],
                  });
                }}
              >
                {/* Background Cover Image */}
                <img
                  src={hero.image}
                  alt={hero.title}
                  className="absolute inset-0 w-full h-full object-cover group-hover:scale-105 transition-transform duration-700 brightness-[0.88]"
                />

                {/* Gradient Overlays */}
                <div className="absolute inset-0 bg-gradient-to-t from-black/75 via-black/25 to-transparent" />
                <div className="absolute inset-0 bg-gradient-to-r from-black/60 via-transparent to-transparent" />

                {/* Banner Content */}
                <div className="absolute inset-0 p-6 sm:p-8 flex flex-col justify-between">
                  <div className="max-w-md">
                    <h2 className="text-2xl sm:text-3xl font-extrabold text-white leading-tight drop-shadow-sm whitespace-pre-line tracking-tight">
                      {hero.title}
                    </h2>
                  </div>

                  <div className="flex items-center gap-3">
                    <button
                      onClick={(e) => copyPromptText(hero.id, hero.prompt, e)}
                      className="flex items-center gap-2.5 px-4 sm:px-5 py-2 rounded-full bg-black/85 hover:bg-black backdrop-blur-md text-white text-xs sm:text-sm font-semibold border border-white/15 transition-all shadow-lg active:scale-95"
                    >
                      <span className="w-4 h-4 rounded-full bg-white flex items-center justify-center shrink-0">
                        <Play className="w-2 h-2 fill-slate-950 text-slate-950 ml-0.5" />
                      </span>
                      <span>{copiedId === hero.id ? "Copied Prompt!" : hero.buttonText}</span>
                    </button>

                    <button
                      onClick={(e) => {
                        e.stopPropagation();
                        copyPromptText(hero.id, hero.prompt);
                      }}
                      className="p-2 rounded-full bg-white/10 hover:bg-white/20 backdrop-blur-md text-white border border-white/15 transition"
                      title="Quick Copy"
                    >
                      {copiedId === hero.id ? (
                        <Check className="w-3.5 h-3.5 text-emerald-400" />
                      ) : (
                        <Copy className="w-3.5 h-3.5" />
                      )}
                    </button>
                  </div>
                </div>
              </div>
            ))}
          </section>

          {/* ── 3. Category Filter Pills Row ── */}
          <div className="relative z-10 flex items-center gap-2.5 sm:gap-3 overflow-x-auto pb-5 scrollbar-none">
            {CATEGORIES.map((cat) => {
              const Icon = cat.icon;
              const isActive = activeCategory === cat.id;
              return (
                <button
                  key={cat.id}
                  onClick={() => setActiveCategory(cat.id)}
                  className={`flex items-center gap-2 px-5 py-2.5 rounded-[18px] text-xs sm:text-sm whitespace-nowrap transition-all duration-200 border ${
                    isActive
                      ? "bg-white/35 text-white border-white/50 shadow-md font-semibold backdrop-blur-xl"
                      : "bg-white/12 text-white/80 border-white/15 hover:bg-white/20 hover:text-white backdrop-blur-md font-medium"
                  }`}
                >
                  <Icon className={`w-4 h-4 ${isActive ? "text-white" : "text-white/70"}`} />
                  <span>{cat.label}</span>
                </button>
              );
            })}
          </div>

          {/* ── 4. Section Title & View Action Buttons ── */}
          <div className="relative z-10 flex items-center justify-between pt-1 pb-4">
            <div className="flex items-center gap-3">
              <h3 className="text-xl sm:text-2xl font-bold text-white capitalize tracking-tight">
                Trending in {CATEGORIES.find((c) => c.id === activeCategory)?.label || "Prompts"}
              </h3>
              <span className="hidden sm:inline-block px-2.5 py-0.5 rounded-full bg-white/10 text-white/70 text-xs font-medium border border-white/10">
                {filteredItems.length} prompts
              </span>
            </div>

            <div className="flex items-center gap-2">
              <div className="flex items-center bg-black/55 backdrop-blur-md border border-white/15 rounded-full px-3 py-1.5 gap-2.5 text-white/70">
                <button
                  type="button"
                  title="Filter options"
                  className="hover:text-white transition"
                >
                  <SlidersHorizontal className="w-3.5 h-3.5" />
                </button>
                <span className="w-px h-3 bg-white/20" />
                <button
                  type="button"
                  title="Sort order"
                  className="hover:text-white transition"
                >
                  <Filter className="w-3.5 h-3.5" />
                </button>
              </div>
            </div>
          </div>

          {/* ── 5. Poster Card Grid (6 Cards Row matching screenshot) ── */}
          <div className="relative z-10 grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 lg:grid-cols-6 gap-4 sm:gap-5">
            {filteredItems.map((item) => (
              <div
                key={item.id}
                onClick={() => setSelectedPrompt(item)}
                className="group cursor-pointer flex flex-col"
              >
                {/* Poster Cover */}
                <div className="relative aspect-[3/4] w-full rounded-[22px] overflow-hidden bg-slate-900 border border-white/15 shadow-md group-hover:border-white/40 transition-all duration-300">
                  <img
                    src={item.image}
                    alt={item.title}
                    className="w-full h-full object-cover group-hover:scale-105 transition-transform duration-500"
                  />

                  {/* Hover Quick Copy Overlay */}
                  <div className="absolute inset-0 bg-black/50 opacity-0 group-hover:opacity-100 transition-opacity duration-200 flex items-center justify-center p-3 gap-2 backdrop-blur-xs">
                    <button
                      onClick={(e) => copyPromptText(item.id, item.prompt, e)}
                      className="p-3 bg-white text-slate-950 rounded-full shadow-xl transition transform hover:scale-110 active:scale-95 font-bold"
                      title="Quick Copy Prompt"
                    >
                      {copiedId === item.id ? (
                        <Check className="w-4 h-4 text-emerald-600" />
                      ) : (
                        <Copy className="w-4 h-4" />
                      )}
                    </button>
                    <button
                      onClick={(e) => toggleBookmark(item.id, e)}
                      className="p-3 bg-white/20 hover:bg-white/30 text-white rounded-full backdrop-blur-md border border-white/20 transition transform hover:scale-110"
                      title="Bookmark"
                    >
                      <Bookmark
                        className={`w-4 h-4 ${
                          bookmarkedIds.has(item.id) ? "fill-amber-400 text-amber-400" : ""
                        }`}
                      />
                    </button>
                  </div>
                </div>

                {/* Card Title and Rating strictly matching screenshot */}
                <div className="mt-2.5 px-0.5">
                  <h4 className="text-xs sm:text-sm font-semibold text-white/95 line-clamp-1 group-hover:text-white transition">
                    {item.title}
                  </h4>
                  <div className="flex items-center gap-2 mt-0.5 text-[11px] text-white/60 font-medium">
                    <span className="flex items-center gap-1 text-amber-400 font-semibold">
                      ★ {item.rating}
                    </span>
                    <span>{item.year}</span>
                  </div>
                </div>
              </div>
            ))}
          </div>

          {/* Empty State */}
          {filteredItems.length === 0 && (
            <div className="py-20 text-center text-slate-400">
              <Layers className="w-12 h-12 mx-auto text-slate-500 mb-3 opacity-60" />
              <p className="text-base font-semibold text-slate-200">No prompts found</p>
              <p className="text-xs text-slate-400 mt-1">Try switching categories or clearing search keywords</p>
            </div>
          )}

          {/* ── 6. Prompt Details Modal Popup ── */}
          {selectedPrompt && (
            <div
              className="fixed inset-0 z-50 flex items-center justify-center bg-black/80 backdrop-blur-md p-4 animate-in fade-in duration-200"
              onClick={() => setSelectedPrompt(null)}
            >
              <div
                className="relative w-full max-w-2xl bg-[#1d2736] border border-white/20 rounded-3xl p-6 sm:p-8 shadow-2xl overflow-hidden"
                onClick={(e) => e.stopPropagation()}
              >
                <button
                  onClick={() => setSelectedPrompt(null)}
                  className="absolute top-5 right-5 p-2 rounded-full bg-white/5 hover:bg-white/10 text-slate-400 hover:text-white transition"
                >
                  <X className="w-5 h-5" />
                </button>

                <div className="flex gap-4 items-start">
                  <img
                    src={selectedPrompt.image}
                    alt={selectedPrompt.title}
                    className="w-20 h-28 object-cover rounded-2xl border border-white/15 shadow-md shrink-0"
                  />
                  <div>
                    <span className="inline-block px-3 py-0.5 rounded-full text-[10px] font-bold bg-cyan-500/20 text-cyan-300 border border-cyan-400/30 mb-1.5">
                      {selectedPrompt.model}
                    </span>
                    <h3 className="text-lg sm:text-xl font-bold text-white">
                      {selectedPrompt.title}
                    </h3>
                    <div className="flex items-center gap-3 mt-1 text-xs text-slate-400 font-medium">
                      <span className="text-amber-400">★ {selectedPrompt.rating}</span>
                      <span>•</span>
                      <span>{selectedPrompt.copies || selectedPrompt.year} uses</span>
                      <span>•</span>
                      <span className="capitalize">{selectedPrompt.category}</span>
                    </div>
                    <p className="text-xs sm:text-sm text-slate-300 mt-2 line-clamp-2">
                      {selectedPrompt.description}
                    </p>
                  </div>
                </div>

                {/* Prompt Box */}
                <div className="mt-6">
                  <div className="flex items-center justify-between mb-2">
                    <span className="text-xs font-bold text-slate-400 uppercase tracking-wider">
                      Full Production Prompt
                    </span>
                    <button
                      onClick={() => copyPromptText("modal-copy", selectedPrompt.prompt)}
                      className="flex items-center gap-1.5 text-xs font-semibold text-cyan-400 hover:text-cyan-300 transition"
                    >
                      {copiedId === "modal-copy" ? (
                        <>
                          <Check className="w-3.5 h-3.5 text-emerald-400" /> Copied!
                        </>
                      ) : (
                        <>
                          <Copy className="w-3.5 h-3.5" /> Copy Prompt
                        </>
                      )}
                    </button>
                  </div>

                  <div className="p-4 bg-black/50 border border-white/10 rounded-2xl text-xs sm:text-sm font-mono text-slate-200 select-all leading-relaxed max-h-48 overflow-y-auto">
                    {selectedPrompt.prompt}
                  </div>
                </div>

                {/* Parameters & Tags */}
                {selectedPrompt.parameters && (
                  <div className="mt-3 flex items-center gap-2 text-[11px] text-slate-400 font-mono">
                    <span className="text-cyan-400 font-semibold">Parameters:</span>
                    <span className="bg-white/5 px-2.5 py-1 rounded-md border border-white/5">
                      {selectedPrompt.parameters}
                    </span>
                  </div>
                )}

                <div className="flex flex-wrap gap-2 mt-4">
                  {selectedPrompt.tags?.map((tag) => (
                    <span
                      key={tag}
                      className="px-3 py-1 rounded-full text-[11px] bg-white/5 border border-white/10 text-slate-300"
                    >
                      #{tag}
                    </span>
                  ))}
                </div>

                {/* Footer Actions */}
                <div className="mt-6 flex items-center justify-between pt-4 border-t border-white/10">
                  <button
                    onClick={() => toggleBookmark(selectedPrompt.id)}
                    className="flex items-center gap-2 text-xs font-medium text-slate-400 hover:text-white transition"
                  >
                    <Bookmark
                      className={`w-4 h-4 ${
                        bookmarkedIds.has(selectedPrompt.id) ? "fill-cyan-400 text-cyan-400" : ""
                      }`}
                    />
                    {bookmarkedIds.has(selectedPrompt.id) ? "Saved" : "Save Prompt"}
                  </button>

                  <div className="flex items-center gap-3">
                    <button
                      onClick={() => setSelectedPrompt(null)}
                      className="px-5 py-2.5 rounded-xl bg-white/5 hover:bg-white/10 text-xs sm:text-sm font-medium text-slate-300 transition"
                    >
                      Close
                    </button>
                    <button
                      onClick={() => copyPromptText("modal-btn", selectedPrompt.prompt)}
                      className="px-6 py-2.5 rounded-xl bg-cyan-500 hover:bg-cyan-400 text-slate-950 text-xs sm:text-sm font-bold transition shadow-lg flex items-center gap-2 active:scale-95"
                    >
                      {copiedId === "modal-btn" ? (
                        <>
                          <Check className="w-4 h-4 text-slate-950" /> Copied to Clipboard
                        </>
                      ) : (
                        <>
                          <Copy className="w-4 h-4 text-slate-950" /> Copy & Use
                        </>
                      )}
                    </button>
                  </div>
                </div>
              </div>
            </div>
          )}
        </div>
      </div>

      {/* ── 7. Converto Modern Rounded Card Footer Matching Uploaded Image ── */}
      <div className="w-full bg-[#0a0c10] py-4">
        <ConvertoFooter
          brandName="CONVERTO"
          displayWordmark="CONVRTO"
          description="Reneria is a full-service interior design agency specializing in residential and commercial design."
          copyrightText="©2025 Converto All rights reserved."
          creditText="Design by Grabui - Powered by Placeholder"
        />
      </div>
    </div>
  );
}
